# AI-Powered Smart Retail & Customer Intelligence Platform
# Multi-stage build: train models at build time, then ship a slim runtime image.

FROM python:3.12-slim AS builder

WORKDIR /build

RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    libgl1 \
    libglib2.0-0 \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir --user -r requirements.txt

COPY app ./app
COPY data ./data

# Train all three ML/DL models during the image build so the runtime
# container starts with serialized artifacts already in models/
RUN mkdir -p models \
    && python app/modules/vision/generate_product_data.py \
    && cd app/modules/vision && python train_product_classifier.py && cd /build \
    && python -m app.modules.nlp.train_sentiment \
    && python -m app.modules.chatbot.train_intent

# ---- Runtime stage ----
FROM python:3.12-slim

WORKDIR /app

RUN apt-get update && apt-get install -y --no-install-recommends \
    libgl1 \
    libglib2.0-0 \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Copy installed Python packages and trained model artifacts from the builder
COPY --from=builder /root/.local /root/.local
COPY --from=builder /build/models ./models
COPY --from=builder /build/data ./data
COPY app ./app

ENV PATH=/root/.local/bin:$PATH \
    PYTHONUNBUFFERED=1

EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=5s --start-period=15s --retries=3 \
    CMD curl -f http://localhost:8000/health || exit 1

CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
